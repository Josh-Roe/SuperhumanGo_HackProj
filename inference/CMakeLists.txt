cmake_minimum_required(VERSION 3.20)
project(superhuman_onnx_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted official ONNX Runtime CPU distribution")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_executable(onnx_runner main.cpp)
target_include_directories(onnx_runner PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(onnx_runner PRIVATE "${ONNXRUNTIME_LIBRARY}")
if(MSVC)
  target_compile_options(onnx_runner PRIVATE /W4)
else()
  target_compile_options(onnx_runner PRIVATE -Wall -Wextra -Wpedantic)
endif()
